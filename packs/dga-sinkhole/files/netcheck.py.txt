# netcheck.py - the domain generation routine, recovered from a packed loader
import datetime

SEED = 0x2F1C9
TLDS = [".top", ".xyz", ".info", ".biz"]


def gen(d):
    s = SEED ^ (d.year * 10000 + d.month * 100 + d.day)
    name = ""
    for _ in range(12):
        s = (s * 1103515245 + 12345) & 0x7FFFFFFF
        name += chr(97 + (s >> 16) % 26)
    return name + TLDS[s % 4]


def rendezvous():
    return gen(datetime.date.today())
